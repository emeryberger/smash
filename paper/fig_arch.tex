\centering
\begin{tikzpicture}[
  state/.style={draw, rounded corners=3pt, font=\footnotesize\ttfamily,
                minimum width=1.5cm, minimum height=0.5cm,
                fill=black!5, thick},
  forward/.style={draw, ->, >=stealth, semithick},
  fault/.style={draw, ->, >=stealth, semithick, dashed, color=black!50},
  phaselbl/.style={fill=white, inner sep=1pt, font=\scriptsize\bfseries,
                   draw, circle, minimum size=4.5mm},
  elbl/.style={font=\scriptsize, fill=white, inner sep=0.5pt},
]
  % Layout: two rows, forward path goes clockwise
  %
  %    EMPTY ---alloc--> ACTIVE ---[3]--> MONITORING
  %                         ^                 |
  %                     faults             [2]|
  %                         |                 v
  %    COMPRESSED <--ROI-- COMPRESSING
  %
  \node[state] (empty) {EMPTY};
  \node[state, right=1.1cm of empty] (active) {ACTIVE};
  \node[state, right=1.1cm of active, minimum width=1.9cm] (monitor) {MONITORING};

  \node[state, below=1.3cm of monitor] (compressing) {COMPRESSING};
  \node[state, below=1.3cm of empty, minimum width=2.0cm] (compressed)
    {\shortstack{COMPRESSED\\[-2pt]\scriptsize\rmfamily(zstd-1/zstd-9)}};

  % ── Forward path (solid) ─────────────────────────────────────

  % alloc
  \draw[forward] (empty) -- node[elbl, above] {alloc} (active);

  % Phase 3: ACTIVE → MONITORING
  \draw[forward] (active) -- node[phaselbl] {3} (monitor);

  % Phase 1: self-loop on MONITORING
  \draw[forward] (monitor) edge[loop above, looseness=5, min distance=6mm]
    node[phaselbl, above=0.05cm] {1} (monitor);

  % Phase 2: MONITORING → COMPRESSING
  \draw[forward] (monitor) -- node[phaselbl] {2} (compressing);

  % ROI: COMPRESSING → COMPRESSED
  \draw[forward] (compressing) -- node[elbl, below] {ROI} (compressed);

  % free: ACTIVE → EMPTY (only when entire span is empty)
  \draw[forward] (active) edge[bend left=25]
    node[elbl, above] {\shortstack{span\\[-2pt]empty}} (empty);

  % ── Fault paths (dashed, back to ACTIVE) ─────────────────────

  % MONITORING → ACTIVE  (write fault, PROT_READ)
  \draw[fault] (monitor) edge[bend left=20]
    node[elbl, below, text=black!50] {fault\,(W)} (active);

  % COMPRESSING → ACTIVE: unreachable (compressor holds per-page lock
  % throughout COMPRESSING; verified via TLA+ model checking).

  % COMPRESSED → ACTIVE  (any fault, decompress)
  \draw[fault] (compressed) --
    node[elbl, left=0.05cm, text=black!50] {fault\,(RW)} (active);

\end{tikzpicture}
\caption{\textbf{Page state machine.}
  Solid arrows: compressor-thread transitions.
  Circled numbers are the three phases of each tick:
  \textcircled{\small 1}~update cold counts,
  \textcircled{\small 2}~compress cold pages (algorithm by ROI),
  \textcircled{\small 3}~set remaining active pages read-only.
  Dashed arrows: signal-handler fault transitions back to
  \textsc{active}.}
\label{fig:arch}
\end{figure}
